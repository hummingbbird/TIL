# 정렬 알고리즘 2탄: merge sort, quick sort, heap sort

## 🙏 Merge Sort

- 분할 정복 알고리즘 → **나눠서**, 각각 **정렬**하고, **합친다** → **재귀적** 성격을 지님
- T(n) = 2T(n/2)+cn → **Θ(nlogn), out-of-place, stable**
- index 기준으로 코드가 진행됨
- Θ(nlogn), out-of-place, stable

```python
# 한 개의 array로 합치는 함수
# 계속 i, j값 바꿔가며 비교해서 넣음
# 그러다가 둘 중의 하나가 index를 벗어나는 순간 남은거 쭉 넣어주고 마무리
def merge(arr,l, m, r):
  tmp = arr[l:r] # 현재 arr 형태 따로 저장해둠
  # i, j는 각각 분할되어있는 array의 index=0 위치, k는 pointer 같은 아이
  i, j, k = 0, m-l, l
  
  # m-l, r-l = 분할된 arr 두 개의 마지막 index (index의 한도선)
  while i < (m-l) and j < (r-l): # i와 j중 하나가 out of range가 되면 종료
    if tmp[i] < tmp[j]:
      arr[k] = tmp[i]
      i+=1
    else:
      arr[k] = tmp[j]
      j+=1
    k+=1
  while i < (m-l): # 1<2: i가 index의 한도선을 넘어갈 때까지 반복복
    arr[k] = tmp[i]
    i+=1
    k+=1
  while j < (r-l): # 3<4: j가 index의 한도선을 넘어갈 때까지 반복
    arr[k] = tmp[j]
    j+=1
    k+=1

# 재귀로 구현
def mergesort(arr, l, r):
	if l+1 == r: return # 종료조건: arr가 한 블록이 되면
	m = (l+r)//2 # 중심인 m값 갱신s
	mergesort(arr, l, m)
	mergesort(arr, m, r)
	merge(arr, l, m, r)
```

## 💨 Quicksort

- merge sort와 비슷한 분할 정복 알고리즘
- **Pivot**을 기준으로 영역을 나누어 각각 정렬함(보통 pivot은 맨 뒤)
- Θ(n logn) ,in-place, unstable

```python
# 분할 함수: pivot을 기준으로 값들을 왼쪽 오른쪽으로 나눔
def partition(arr, l, r):
  p = arr[r-1] # pivot은 제일 뒤에 거
  i = l #i=0

  for j in range(l, r-1): # 처음~마지막-1까지 반복(p가 마지막이니까)
    if arr[j] <= p: # p보다 값이 작거나 같으면 i <-> j
      arr[j], arr[i] = arr[i], arr[j]
      i+=1
  
  arr[i], arr[r - 1] = arr[r - 1], arr[i]
  return i
  
  def partition11(arr, l, r):
```

```python
def quicksort(arr, l, r):
```

+) 다른 종류의 quick sort 함수

```python
def partition2(A, low, high):
    pivot_idx = low  # 피벗의 인덱스를 임의의 값으로 설정 (현재는 제일 왼쪽 값으로 설정)
    pivot = A[pivot_idx]

    A[pivot_idx], A[high] = A[high], A[pivot_idx]  # 비교를 위해 피벗을 가장 오른쪽 값과 swap

    i = low
    for j in range(low, high):
        if A[j] < pivot:  # 인덱스 j는 피벗보다 작은 값을 찾아서, A[i]와 swap
            A[i], A[j] = A[j], A[i]
            i += 1

    pivot_idx = i  # 반복문을 마친 배열에서 i는 피벗의 인덱스가 되어야 함.
    A[pivot_idx], A[high] = A[high], A[pivot_idx]

    return pivot_idx

def quicksort2(A, low, high):
    if low < high:
        pivot_idx = partition2(A, low, high)  # 분할 함수를 진행하여 피벗을 기준으로 분할을 진행
        quicksort2(A, low, pivot_idx - 1)  # 분할의 왼쪽 부분에 대해서 재귀
        quicksort2(A, pivot_idx + 1, high)  # 분할의 오른쪽에 대해서 재귀
```

## 🌲 Heapsort

- heap 자료구조의 형태를 따르는 알고리즘
- max와 j를 swap하고(**find_max**) 재정렬(**sift down**)하는 과정의 반복
- siftdown 핵심: 노드 두 자식이 heap이면 자기까지 heap으로 만들어 → Θ(nlogn)
- build_heap: leaf node가 아닌, 가장 오른쪽부터 순서대로 siftdown 진행 → Θ(n)
- 시간 복잡도: Θ(n) + Θ(nlogn) = Θ(nlogn)
- Θ(n logn) ,in-place, stable(단, 코드에 따라 달라질 수 있음)

```python
def heap_sort(arr):
    build_heap(arr) #일단 heap 형태로 만듦
    n=len(arr)

    for i in range(n-1, 0, -1): #뒤에서부터 반복
       arr[0], arr[i] = arr[i], arr[0] #맨앞(max)<->i랑 swap
       siftdown(0,arr,i) #sift down 진행(남은 node 다시 heap 형태로 만들어줌)

def siftdown(i, arr, n):
   c = 2*i + 1 # c: 왼쪽 노드 
   
   #재귀 없이 구현
   while c<n:
      if c+1<n and arr[c] <arr[c+1]:
         c+=1
      if arr[i] > arr[c]:
         break
      arr[i], arr[c] = arr[c], arr[i]
      i=c
      c=2*i+1   

# case2: 재귀 사용해서 구현
def siftdown2(i, arr, n):
    c = 2 * i + 1  # 왼쪽 자식 노드
    if c >= n:  # 자식 노드가 없는 경우 종료
        return
    
    if c + 1 < n and arr[c] < arr[c + 1]:  # 오른쪽 자식이 더 크면 오른쪽 자식 선택
        c += 1
    
    if arr[i] >= arr[c]:  # 부모가 자식보다 크거나 같으면 종료
        return
    
    # 부모와 자식 교환
    arr[i], arr[c] = arr[c], arr[i]
    
    # 재귀적으로 자식 노드에 대해 siftdown 호출
    siftdown(c, arr, n)
    
def build_heap(arr):
   n = len(arr)
   for i in range(n//2-1, -1, -1):
      siftdown2(i, arr, n)
```

## +) min-heap을 내림차순 heapsort

```python
# 내림차순 정렬을 위한 min heap 기반 heap sort
def heap_sort(arr):
    build_heap(arr)  # 일단 min heap 형태로 만들어
    print(f"빌드 힙 결과: {arr}")
    n = len(arr)

    for i in range(n - 1, 0, -1):  # 뒤에서부터 반복
        arr[0], arr[i] = arr[i], arr[0]  # 맨 앞(min) <-> i와 swap
        siftdown(0, arr, i)  # sift down 진행 (남은 노드를 다시 heap 형태로 만들어줌)
        print(f"siftdown한 결과: {arr}")

# siftdown 함수 (min heap 유지)
def siftdown(i, arr, n):
    c = 2 * i + 1  # 왼쪽 자식 노드
    
    # 재귀 없이 구현
    while c < n:
        if c + 1 < n and arr[c] > arr[c + 1]:  # 오른쪽 자식이 더 작으면 오른쪽 자식 선택
            c += 1
        if arr[i] < arr[c]:  # 부모가 자식보다 작으면 break (min heap 조건)
            break
        arr[i], arr[c] = arr[c], arr[i]  # 부모와 자식 교환
        i = c
        c = 2 * i + 1

# case2: 재귀 사용한 siftdown 구현 (min heap 유지)
def siftdown2(i, arr, n):
    c = 2 * i + 1  # 왼쪽 자식 노드
    if c >= n:  # 자식 노드가 없는 경우 종료
        return
    
    if c + 1 < n and arr[c] > arr[c + 1]:  # 오른쪽 자식이 더 작으면 오른쪽 자식 선택
        c += 1
    
    if arr[i] <= arr[c]:  # 부모가 자식보다 작으면 종료 (min heap 조건)
        return
    
    # 부모와 자식 교환
    arr[i], arr[c] = arr[c], arr[i]
    
    # 재귀적으로 자식 노드에 대해 siftdown 호출
    siftdown(c, arr, n)

# min heap을 만들어주는 build_heap 함수
def build_heap(arr):
    n = len(arr)
    for i in range(n // 2 - 1, -1, -1):
        siftdown(i, arr, n)

arr = [9,8,7,6,5,4,4,3,2,1]

print("min heap을 이용한 내림차순 정렬")
heap_sort(arr)
```
